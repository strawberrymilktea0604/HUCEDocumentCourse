#!/bin/bash
if [ "$#" -lt 2 ]; then
  echo "Bạn chưa nhập tham số!"
  exit 1
fi
file="$1"
directory="$2"
found=$(find "$directory" -name "$file" | wc -l)
if [ "$found" -gt 0 ]; then
  echo "File có tồn tại trong thư mục $directory"
else
  echo "File không tồn tại trong thư mục $directory"
fi
