#!/bin/bash
if [ "$#" -ne 2 ]; then
  echo "Tham so khong hop le, nhap lai!"
  exit 1
fi
thang="$1"
nam="$2"
case "$thang" in
  1|3|5|7|8|10|12)
    days=31
    ;;
  4|6|9|11)
    days=30
    ;;
  2)
    if (((nam % 400 == 0) || (nam % 4 == 0 && nam % 100 != 0) )); then
      days=29
    else
      days=28
    fi
    ;;
  *)
    echo "Thang khong hop le: $thang"
    exit 1
    ;;
esac
echo "Thang $thang cua nam $nam co $days ngay"
