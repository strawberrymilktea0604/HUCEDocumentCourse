#!/bin/bash
hinh=$1
n=$2
m=$3
case $hinh in
  chu_nhat)
    if [ -z "$n" ] || [ -z "$m" ]; then
      echo "Chua du tham so cho hinh chu nhat"
      exit 1
    fi
    for ((i=1; i<=n; i++)); do
      for ((j=1; j<=m; j++)); do
        echo -n "*"
      done
      echo
    done
    ;;
  tam_giac_duoi)
    if [ -z "$n" ]; then
      echo "Chua du tham so cho tam giac duoi"
      exit 1
    fi
    for ((i=1; i<=n; i++)); do
      for ((j=1; j<=i; j++)); do
        echo -n "*"
      done
      echo
    done
    ;;
  tam_giac_tren)
    if [ -z "$n" ]; then
      echo "Chua du tham so cho tam giac tren"
      exit 1
    fi
    for ((i=n; i>=1; i--)); do
      for ((j=1; j<=i; j++)); do
        echo -n "*"
      done
      echo
    done
    ;;
  tam_giac_so)
    if [ -z "$n" ]; then
      echo "Chua du tham so cho tam giac so"
      exit 1
    fi
    num=1
    for ((i=1; i<=n; i++)); do
      for ((j=1; j<=i; j++)); do
        echo -n "$num "
        num=$((num+1))
      done
      echo
    done
    ;;
  *)
    echo "Chuong trinh chua biet ve hinh nay"
    ;;
esac
