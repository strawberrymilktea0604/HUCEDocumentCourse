#!/bin/bash
if [ "$#" -eq 0 ]; then
  echo "Bạn chưa nhập chiều cao tam giác!"
  exit 1
fi

n="$1"

for ((i=n; i>=2; i--)); do
  # In khoảng trắng (căn giữa)
  for ((s=i; s<n; s++)); do
    printf " "
  done

  # In dấu sao hoặc khoảng trống (rỗng bên trong)
  for ((j=1; j<=2*i-1; j++)); do
    # Dòng đáy hoặc cạnh ngoài
    if [[ $i -eq n || $j -eq 1 || $j -eq $((2*i-1)) ]]; then
      printf "*"
    else
      printf " "
    fi
  done

  echo
done
