#!/bin/bash
if [ "$#" -eq 0 ]; then
  echo "Bạn chưa nhập tham số!"
  exit 1
fi

n="$1"

# Bắt đầu từ i = 2 để bỏ đỉnh
for ((i=2; i<=n; i++)); do
  for ((j=1; j<=i; j++)); do
    if [[ $i -eq n || $j -eq 1 || $j -eq i ]]; then
      printf "*"
    else
      printf " "
    fi
  done
  echo
done
