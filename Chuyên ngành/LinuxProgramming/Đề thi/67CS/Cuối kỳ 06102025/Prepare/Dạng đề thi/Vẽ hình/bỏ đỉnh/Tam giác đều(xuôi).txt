#!/bin/bash
if [ "$#" -eq 0 ]; then
  echo "Bạn chưa nhập chiều cao tam giác!"
  exit 1
fi

n="$1"

for ((i=2; i<=n; i++)); do   # bắt đầu từ 2 thay vì 1
  # Căn trái bằng khoảng trắng
  for ((s=i; s<n; s++)); do
    printf " "
  done

  # In sao hoặc khoảng trống (tam giác rỗng)
  for ((j=1; j<=i; j++)); do
    # Viền ngoài hoặc đáy
    if [[ $i -eq n || $j -eq 1 || $j -eq i ]]; then
      printf "* "
    else
      printf "  "
    fi
  done

  echo
done
