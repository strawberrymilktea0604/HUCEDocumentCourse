#!/bin/bash
if [ "$#" -lt 1 ]; then
  echo "Bạn chưa nhập độ dài cạnh hình vuông!"
  exit 1
fi

canh="$1"

for ((i=1; i<=canh; i++)); do
  for ((j=1; j<=canh; j++)); do
    # In viền ngoài (hàng đầu, hàng cuối, cột đầu, cột cuối)
    if [[ $i -eq 1 || $i -eq $canh || $j -eq 1 || $j -eq $canh ]]; then
      printf "*"
    else
      printf " "
    fi
  done
  echo
done
