#!/bin/bash
if [ "$#" -eq 0 ]; then
  echo "Bạn chưa nhập chiều cao tam giác!"
  exit 1
fi

n="$1"

# Bỏ đỉnh (hàng i=1)
for ((i=2; i<=n; i++)); do
  # In khoảng trắng để căn phải
  for ((s=i; s<n; s++)); do
    printf "  "
  done

  # In sao hoặc khoảng trống bên trong
  for ((j=1; j<=i; j++)); do
    if [[ $i -eq n || $j -eq 1 || $j -eq i ]]; then
      printf "* "
    else
      printf "  "
    fi
  done

  echo
done
