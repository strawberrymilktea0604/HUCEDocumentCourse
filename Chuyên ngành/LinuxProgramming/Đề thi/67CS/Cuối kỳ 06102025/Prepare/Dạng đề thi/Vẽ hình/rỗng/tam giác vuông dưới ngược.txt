#!/bin/bash
if [ "$#" -eq 0 ]; then
  echo "Bạn chưa nhập tham số!"
  exit 1
fi

n="$1"

for ((i=n; i>=1; i--)); do
  for ((j=1; j<=i; j++)); do
    # In viền: dòng đầu, dòng cuối, hoặc cạnh trái / phải
    if [[ $i -eq n || $i -eq 1 || $j -eq 1 || $j -eq i ]]; then
      printf "*"
    else
      printf " "
    fi
  done
  echo
done
