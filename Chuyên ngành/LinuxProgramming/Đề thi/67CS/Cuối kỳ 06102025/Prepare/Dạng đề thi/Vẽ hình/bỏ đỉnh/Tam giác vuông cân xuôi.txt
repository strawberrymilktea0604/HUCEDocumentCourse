#!/bin/bash
if [ "$#" -eq 0 ]; then
  echo "Bạn chưa nhập chiều cao tam giác!"
  exit 1
fi

n="$1"

# Bỏ đỉnh (hàng i=1)
for ((i=2; i<=n; i++)); do
  for ((j=1; j<=i; j++)); do
    # In sao ở viền ngoài (cạnh trái, cạnh chéo, hoặc đáy)
    if [[ $i -eq n || $j -eq 1 || $j -eq i ]]; then
      printf "* "
    else
      printf "  "
    fi
  done
  echo
done
