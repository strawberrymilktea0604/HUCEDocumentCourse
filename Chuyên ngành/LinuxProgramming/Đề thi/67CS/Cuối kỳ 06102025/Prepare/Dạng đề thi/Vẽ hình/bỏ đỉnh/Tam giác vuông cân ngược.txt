#!/bin/bash
if [ "$#" -eq 0 ]; then
  echo "Bạn chưa nhập chiều cao tam giác!"
  exit 1
fi

n="$1"

# Bỏ đỉnh (hàng đầu), vẫn giữ đáy
for ((i=n-1; i>=1; i--)); do
  for ((j=1; j<=i; j++)); do
    # Viền ngoài (dòng đầu, cạnh trái, cạnh chéo)
    if [[ $i -eq 1 || $j -eq 1 || $j -eq i ]]; then
      printf "* "
    else
      printf "  "
    fi
  done
  echo
done
